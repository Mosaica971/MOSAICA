
*ooooooo Calcul de la DT50 moyenne par culture pondérée par les quantités de produits phytosanitaires utilisés (en jours/ha/an) moyenne, année de plantation comprise (pour le calcul de R_PEST)

* La persistance dans le sol ou temps de demi-vie (DT50) est évaluée par le temps de dégradation ou la dissipation de 50 % de la substance active présente dans le sol. La base de données « phytosanitaires »
*renseigne la persistance (DT50) mesurée au champ. Elle est exprimée en jours.

         DT50_OTK_Cult(SC,SK)$((        Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=0)*Data_OTK(SK,"DOSE")
                                     + (Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=1)*Data_OTK(SK,"DOSE") )/Duree_Plant_Cult(SC,SH)
                                )/  Duree_Cycle_Cult(SC,SH)*12
                               )
                                =((     Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=0)*Data_OTK(SK,"DOSE")*Data_OTK(SK,"DT50")
                                     + (Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=1)*Data_OTK(SK,"DOSE")*Data_OTK(SK,"DT50") )/Duree_Plant_Cult(SC,SH)
                                )/  Duree_Cycle_Cult(SC,SH)*12 )/
                               ((       Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=0)*Data_OTK(SK,"DOSE")
                                     + (Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=1)*Data_OTK(SK,"DOSE") )/Duree_Plant_Cult(SC,SH)
                               )/   Duree_Cycle_Cult(SC,SH)*12
                               );
         STOCK_DT50_OTK_Cult(SH,SC,SK)= DT50_OTK_Cult(SC,SK);

         DT50_Cult(SC)= sum(SK, DT50_OTK_Cult(SC,SK));
         STOCK_DT50_Cult(SH,SC)= DT50_Cult(SC);


*ooooooo Calcul de la dose maximum admise par un humain moyenne par culture pondérée par les quantités de produits phytosanitaires utilisés (en jours/ha/an) moyenne, année de plantation comprise (pour le calcul de R_PEST)
         ADI_OTK_cult(SC,SK)$((        Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=0)*Data_OTK(SK,"DOSE")
                                    + (Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=1)*Data_OTK(SK,"DOSE") )/Duree_Plant_Cult(SC,SH)
                               )/  Duree_Cycle_Cult(SC,SH)*12
                               )
                               =((     Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=0)*Data_OTK(SK,"DOSE")*Data_OTK(SK,"ADI")
                                    + (Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=1)*Data_OTK(SK,"DOSE")*Data_OTK(SK,"ADI") )/Duree_Plant_Cult(SC,SH)
                                )/  Duree_Cycle_Cult(SC,SH)*12 )/
                              ((       Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=0)*Data_OTK(SK,"DOSE")
                                    + (Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=1)*Data_OTK(SK,"DOSE") )/Duree_Plant_Cult(SC,SH)
                               )/   Duree_Cycle_Cult(SC,SH)*12
                              );
         STOCK_ADI_OTK_cult(SH,SC,SK)= ADI_OTK_cult(SC,SK);

         ADI_Cult(SC)= sum(SK, ADI_OTK_cult(SC,SK));
         STOCK_ADI_Cult(SH,SC)= ADI_Cult(SC);

*ooooooo Calcul de la charge en matière active par culture (en kg de MA/ha/an) moyenne, année de plantation comprise (pour le calcul de R_PEST)
         QMA_ha_OTK_cult(SC,SK)= (  Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=0)*Data_OTK(SK,"DOSE")*Data_OTK(SK,"QMA")
                                 + (Matrice_OTK_Cult(SK, SC)$(Data_OTK(SK,"AMORTI")=1)*Data_OTK(SK,"DOSE")*Data_OTK(SK,"QMA") )/Duree_Plant_Cult(SC,SH)
                                )/ Duree_Cycle_Cult(SC,SH)*12;
         STOCK_QMA_ha_OTK_cult(SH,SC,SK)= QMA_ha_OTK_cult(SC,SK);

         QMA_Ha_Cult(SC)= sum(SK, QMA_ha_OTK_cult(SC,SK));
         STOCK_QMA_Ha_Cult(SH,SC)= QMA_Ha_Cult(SC);


*ooooooo Calcul de R-PEST par parcelle ooooooooooooooooooooooooooooooooooooooooo

         LOOP(SC,
                  IF (QMA_Ha_Cult(SC) >  R_Tixier("SEUIL_DOSE_CULT","DEFAVORABLE"), DOSE_defav(SC)=1;
                  );
                  IF (QMA_Ha_Cult(SC) <  R_Tixier("SEUIL_DOSE_CULT","FAVORABLE"),   DOSE_defav(SC)=0;
                  );
                  IF (QMA_Ha_Cult(SC) <= R_Tixier("SEUIL_DOSE_CULT","DEFAVORABLE") and (QMA_Ha_Cult(SC)>R_Tixier("SEUIL_DOSE_CULT","FAVORABLE")),
                        DOSE_defav(SC)$((R_Tixier("SEUIL_DOSE_CULT","DEFAVORABLE") - R_Tixier("SEUIL_DOSE_CULT","FAVORABLE")))
                     =(QMA_Ha_Cult(SC) - R_Tixier("SEUIL_DOSE_CULT","FAVORABLE")) /((R_Tixier("SEUIL_DOSE_CULT","DEFAVORABLE") - R_Tixier("SEUIL_DOSE_CULT","FAVORABLE")));
                  );
         );
         STOCK_DOSE_defav(SH,SC)= DOSE_defav(SC);

         DOSE_fav(SC)= 1-DOSE_defav(SC);
         STOCK_DOSE_fav(SH,SC)= DOSE_fav(SC);

         RUI_fav(SP)= 1-RUI_defav(SP);

         E_SURF(SP,SC)$

                    min(DOSE_fav(SC),  RUI_defav(SP))*ASSOL_Parc(SP,SC)+

                   (10*min(DOSE_defav(SC),RUI_defav(SP))*ASSOL_Parc(SP,SC))))

                  ((min(DOSE_fav(SC),  RUI_fav(SP))  *ASSOL_Parc(SP,SC)+

                    min(DOSE_defav(SC),RUI_defav(SP))*ASSOL_Parc(SP,SC)));

                      DT50_defav(SC)$((R_Tixier("SEUIL_DT50_CULT","DEFAVORABLE") - R_Tixier("SEUIL_DT50_CULT","FAVORABLE")))

                  ;)

                     ((min(ADI_Cult(SC),AQUATOX_Cult(SC))) >  R_Tixier("SEUIL_MIN_ADIAQUATOX","FAVORABLE")),

*                  = ((DT50_Cult(SC) - R_Tixier("SEUIL_MIN_ADIAQUATOX","FAVORABLE"))) / ((R_Tixier("SEUIL_MIN_ADIAQUATOX","DEFAVORABLE") - R_Tixier("SEUIL_MIN_ADIAQUATOX","FAVORABLE")));)

                  /

                  ((0*(min(DOSE_fav(SC),  GUS_fav(SC),  DRAI_fav(SP)))  *ASSOL_Parc(SP,SC)+
                   (3*min(DOSE_fav(SC),   GUS_fav(SC),  DRAI_defav(SP)))*ASSOL_Parc(SP,SC)+

                    min(DOSE_defav(SC),GUS_defav(SC),DRAI_fav(SP))   *ASSOL_Parc(SP,SC)+

                  ADI_defav(SC)$((R_Tixier("SEUIL_ADI_CULT","DEFAVORABLE") - R_Tixier("SEUIL_ADI_CULT","FAVORABLE")))

                  /((R_Tixier("SEUIL_ADI_CULT","DEFAVORABLE") - R_Tixier("SEUIL_ADI_CULT","FAVORABLE")));
                  );
         );

         ADI_fav(SC)= 1-ADI_defav(SC);
         STOCK_ADI_fav(SH,SC)= ADI_fav(SC);


         T_PROF(SP,SC)$
                  ((min(DT50_fav(SC),  ADI_fav(SC))  *ASSOL_Parc(SP,SC)+
                    min(DT50_fav(SC),  ADI_defav(SC))*ASSOL_Parc(SP,SC)+
                    min(DT50_defav(SC),ADI_fav(SC))  *ASSOL_Parc(SP,SC)+
                    min(DT50_defav(SC),ADI_defav(SC))*ASSOL_Parc(SP,SC)))
                  =
                  ((0*(min(DT50_fav(SC),  ADI_fav(SC)))  *ASSOL_Parc(SP,SC)+
                   (4*min(DT50_fav(SC),   ADI_defav(SC)))*ASSOL_Parc(SP,SC)+
                   (6*min(DT50_defav(SC), ADI_fav(SC)))  *ASSOL_Parc(SP,SC)+
                   (10*min(DT50_defav(SC),ADI_defav(SC)))*ASSOL_Parc(SP,SC)))
                  /
                  ((min(DT50_fav(SC),  ADI_fav(SC))   *ASSOL_Parc(SP,SC)+
                    min(DT50_fav(SC),  ADI_defav(SC)) *ASSOL_Parc(SP,SC)+
                    min(DT50_defav(SC),ADI_fav(SC))   *ASSOL_Parc(SP,SC)+
                    min(DT50_defav(SC),ADI_defav(SC)))*ASSOL_Parc(SP,SC));
         STOCK_T_PROF(SH,SP,SC)= T_PROF(SP,SC);


         R_SURF(SP,SC)= ((E_SURF(SP,SC))+(T_SURF(SP,SC)))/2;
         STOCK_R_SURF(SH,SP,SC)= R_SURF(SP,SC);

         R_PROF(SP,SC)= ((E_PROF(SP,SC))+(T_PROF(SP,SC)))/2;
         STOCK_R_PROF(SH,SP,SC)= R_PROF(SP,SC);


         R_PEST(SP,SC)= max(R_SURF(SP,SC), R_PROF(SP,SC));
         STOCK_R_PEST(SH,SP,SC)= R_PEST(SP,SC);

         R_PEST_parc(SP)$SURF_Parc(SP)= (sum(SC, R_PEST(SP,SC)*ASSOL_Parc(SP,SC))) / SURF_Parc(SP);
         STOCK_R_PEST_Parc(SH,SP)= R_PEST_Parc(SP);

         R_PEST_Gwad =  sum(SP$(R_PEST_parc(SP)>6), SURF_Parc(SP));
         STOCK_R_PEST_Gwad(SH)= R_PEST_Gwad;
